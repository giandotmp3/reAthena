-	script	Syn_switch	-1,{
	// 1. matched switch: $@__SW0_VAL is reset to 0 by the case; 2. no match: the switch value is left behind; 3. pre-case statements run
	switch (2) { case 1: mes "one"; break; case 2: mes "two"; break; }
	switch (7) { case 1: mes "no"; break; }
	switch (9) { mes "pre"; case 1: break; default: mes "dflt"; break; }
	// 4. fallthrough into a middle 'default': the compare chain continues with VAL=0 (case 0 matches)
	switch (1) { case 1: mes "one"; default: mes "dflt"; case 0: mes "zero"; break; }
	// 5. nested switch inside a case, leftovers of the inner one
	switch (5) { case 5: switch (6) { case 1: break; } mes "inner"; break; }
	// 6. loop + continue inside a switch inside a loop
	.@i = 0;
	while (.@i < 4) {
		.@i++;
		switch (.@i) { case 1: mes "a"; case 2: mes "b"; break; case 3: continue; default: mes "c"; }
		mes "after" + .@i;
	}
	// 7. switch on a string-returned number and on constants
	switch (getarraysize(.@none)) { case 0: mes "empty"; break; default: mes "nonempty"; }
	switch (3) { case 1: case 2: case 3: mes "123"; break; }
	// 8. leftovers by name (indexes count if/while/do/for/function/switch in source order)
	mes "left=" + $@__sw1_val + "," + $@__sw2_val + "," + $@__sw5_val;
	close;
}

-	script	Syn_expr	-1,{
