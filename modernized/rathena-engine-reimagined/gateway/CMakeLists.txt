cmake_minimum_required(VERSION 3.20)
project(gateway VERSION 0.2.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# GATEWAY_SANITIZE: OFF | ON (= ASan+UBSan) | address | thread
set(GATEWAY_SANITIZE "OFF" CACHE STRING "OFF | ON | address | thread")
option(GATEWAY_WERROR "Treat warnings as errors" ON)
set(GATEWAY_NETPROTO_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../netproto" CACHE PATH "netproto unit (read-only)")
set(GATEWAY_PERSISTENCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../persistence" CACHE PATH "persistence unit (read-only)")

find_package(Threads REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)

set(_warn -Wall -Wextra -Wpedantic)
if(GATEWAY_WERROR)
  list(APPEND _warn -Werror)
endif()
set(_san "")
if(GATEWAY_SANITIZE STREQUAL "ON" OR GATEWAY_SANITIZE STREQUAL "address")
  set(_san -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
elseif(GATEWAY_SANITIZE STREQUAL "thread")
  set(_san -fsanitize=thread -fno-omit-frame-pointer)
endif()

# ---- netproto: compiled read-only from its unit (framing, gating, catalog model, codec set, connect check)
set(NP ${GATEWAY_NETPROTO_DIR})
add_library(gw_netproto STATIC
  ${NP}/src/common.cpp ${NP}/src/spec.cpp ${NP}/src/framing.cpp ${NP}/src/gating.cpp
  ${NP}/src/connect_check.cpp ${NP}/src/codec_set.cpp ${NP}/src/flood_guard.cpp)
target_include_directories(gw_netproto PUBLIC ${NP}/include)
target_compile_features(gw_netproto PUBLIC cxx_std_20)
target_compile_options(gw_netproto PRIVATE -Wall -Wextra -Wpedantic ${_san})
# The catalog of the real client spec is generated by netproto's own codegen, into OUR build tree.
set(GW_GEN_DIR ${CMAKE_CURRENT_BINARY_DIR}/generated)
add_custom_command(
  OUTPUT ${GW_GEN_DIR}/catalog_client_packets.hpp
  COMMAND ${Python3_EXECUTABLE} -I ${NP}/tools/codegen/netproto_codegen.py --proto ${NP}/proto/client_packets.yaml
          --emit-catalog --client re/20211103 --out ${GW_GEN_DIR}
  DEPENDS ${NP}/proto/client_packets.yaml ${NP}/tools/codegen/netproto_codegen.py
  COMMENT "netproto codegen (read-only unit): catalog_client_packets.hpp"
  VERBATIM)
add_custom_target(gw_netproto_generated DEPENDS ${GW_GEN_DIR}/catalog_client_packets.hpp)

# ---- persistence: compiled read-only from its unit
set(PS ${GATEWAY_PERSISTENCE_DIR})
add_library(gw_persist STATIC
  ${PS}/src/domain.cpp ${PS}/src/db_config.cpp ${PS}/src/connection_factory.cpp ${PS}/src/sqlite_connection.cpp
  ${PS}/src/connection_wrappers.cpp ${PS}/src/codec.cpp ${PS}/src/schema.cpp ${PS}/src/stores.cpp ${PS}/src/wal.cpp
  ${PS}/src/dirty_tracker.cpp ${PS}/src/flusher.cpp ${PS}/src/writer.cpp)
target_include_directories(gw_persist PUBLIC ${PS}/include)
target_compile_features(gw_persist PUBLIC cxx_std_20)
target_link_libraries(gw_persist PUBLIC Threads::Threads PRIVATE SQLite::SQLite3)
target_compile_options(gw_persist PRIVATE -Wall -Wextra -Wpedantic ${_san})

# ---- the gateway
add_library(gateway STATIC
  src/crypto.cpp src/hasher.cpp src/rules.cpp src/inproc.cpp src/misc.cpp src/config.cpp
  src/login_service.cpp src/admin_service.cpp src/char_service.cpp
  src/wire.cpp src/packet_handlers.cpp src/runtime.cpp src/loopback.cpp src/persist_adapter.cpp)
add_dependencies(gateway gw_netproto_generated)
target_include_directories(gateway PUBLIC include ${CMAKE_CURRENT_BINARY_DIR})
target_compile_features(gateway PUBLIC cxx_std_20)
target_compile_options(gateway PRIVATE ${_warn})
target_link_libraries(gateway PUBLIC gw_netproto gw_persist Threads::Threads ${CMAKE_DL_LIBS})
if(_san)
  target_compile_options(gateway PUBLIC ${_san})
  target_link_options(gateway PUBLIC ${_san})
endif()

add_executable(rathena_gateway src/main.cpp)
target_link_libraries(rathena_gateway PRIVATE gateway)
target_compile_options(rathena_gateway PRIVATE ${_warn})

enable_testing()
add_executable(gateway_tests
  tests/main.cpp tests/test_pure.cpp tests/test_login.cpp tests/test_char.cpp tests/test_security.cpp
  tests/test_crypto.cpp tests/test_hasher.cpp tests/test_login_ext.cpp tests/test_char_ext.cpp
  tests/test_wire.cpp tests/test_runtime.cpp tests/test_failure_modes.cpp tests/test_persist.cpp)
target_link_libraries(gateway_tests PRIVATE gateway)
target_compile_options(gateway_tests PRIVATE ${_warn})
target_compile_definitions(gateway_tests PRIVATE GATEWAY_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}"
                           GATEWAY_NETPROTO_DIR="${GATEWAY_NETPROTO_DIR}")
add_test(NAME acceptance COMMAND gateway_tests)
