//= Card Seller A-Z
//===== Description: =========================================
//= Sells all cards dropped by mobs, grouped alphabetically.
//= MVP cards are excluded from the list.
//=
//===== Changelogs: ==========================================
//= 1.0 First version [AnnieRuru]

//= 1.3 Built from the item and mob databases instead of SQL tables

prontera,155,177,5	script	Card Seller	100,{
	.@menu$ = getvariableofnpc( .alphabet_menu$, "card_seller_creation" );
	if (.@menu$ == "") {
		mes "[Card Seller]";
		mes "I am sorry, it seems like something went wrong.";
		mes "I cannot find any cards in our database at the moment.";
		mes "Please contact a game master.";
		close;
	}
	mes "[Card Seller]";
	mes "Welcome!";
	mes "I can sell you any normal monster card in the game. Would you like to have a look?";
	next;
	.@s = select(.@menu$) -1;
	close2;
	callshop "card_mob#"+ getvariableofnpc( .alphabet$[.@s], "card_seller_creation" ), 1;
	end;
}

-	script	card_seller_creation	-1,{
	end;
OnInit:
	// Cards dropped (normal drops, not MVP rewards) by any non-MVP monster, grouped by the first
	// letter of their name and sorted by name.
	.@letters$ = "ABCDEFGHIJKLMNOPQRSTUVWXYZ";
	freeloop 1;
	for ( .@mob = 1001; .@mob < 65535; .@mob++ ) {
		if (!getmobdrops(.@mob) || (getmonsterinfo(.@mob, MOB_MODE) & MD_MVP))
			continue;
		.@count = $@MobDrop_count;
		copyarray .@drops[0], $@MobDrop_item[0], .@count;
		for ( .@d = 0; .@d < .@count; .@d++ ) {
			.@item = .@drops[.@d];
			if (.@seen[.@item])
				continue;
			.@seen[.@item] = 1;
			if (getiteminfo(.@item, ITEMINFO_TYPE) != IT_CARD || getiteminfo(.@item, ITEMINFO_SUBTYPE) == CARD_ENCHANT)
				continue;
			.@name$ = getitemname(.@item);
			.@l = strpos(.@letters$, strtoupper(charat(.@name$, 0)));
			if (.@l < 0)
				continue;
			// insertion sort by name into the letter's list (shops hold up to 128 items)
			.@n = getd(".@size_" + .@l);
			if (.@n >= 128)
				continue;
			for ( .@p = .@n; .@p > 0 && getd(".@names_" + .@l + "$[" + (.@p - 1) + "]") > .@name$; .@p-- ) {
				setd ".@names_" + .@l + "$[" + .@p + "]", getd(".@names_" + .@l + "$[" + (.@p - 1) + "]");
				setd ".@ids_" + .@l + "[" + .@p + "]", getd(".@ids_" + .@l + "[" + (.@p - 1) + "]");
			}
			setd ".@names_" + .@l + "$[" + .@p + "]", .@name$;
			setd ".@ids_" + .@l + "[" + .@p + "]", .@item;
			setd ".@size_" + .@l, .@n + 1;
		}
	}
	for ( .@l = 0; .@l < 26; .@l++ ) {
		.@n = getd(".@size_" + .@l);
		if (.@n == 0)
			continue;
		.@letter$ = charat(.@letters$, .@l);
		.alphabet$[.@size_alphabet++] = .@letter$;
		.alphabet_menu$ = .alphabet_menu$ + .@letter$ +" Cards:";
		npcshopdelitem "card_mob#"+ .@letter$, 501;
		for ( .@j = 0; .@j < .@n; .@j++ )
			npcshopadditem "card_mob#"+ .@letter$, getd(".@ids_" + .@l + "[" + .@j + "]"), 1000000;
	}
	freeloop 0;
	end;
}
